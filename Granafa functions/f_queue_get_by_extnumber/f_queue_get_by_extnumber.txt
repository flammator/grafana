DELIMITER $$

USE `tgui`$$

DROP FUNCTION IF EXISTS `f_queue_get_by_extnumber`$$

CREATE DEFINER=`root`@`localhost` FUNCTION `f_queue_get_by_extnumber`(i_queue VARCHAR(50)) RETURNS INT(6)
    READS SQL DATA
    DETERMINISTIC
BEGIN
    DECLARE w3 INT DEFAULT 0;

    SELECT id INTO w3
    FROM queues
    WHERE ext_number = i_queue
    LIMIT 1;

    RETURN w3;
END$$

DELIMITER ;
